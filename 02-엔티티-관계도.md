# 02. 엔티티 관계도

> DDD 아키텍처로 엔티티 간 직접 참조 없이 **ID 참조**만 사용합니다.

## 도메인별 Aggregate 정리

| Aggregate Root | 하위 엔티티 | ID 참조 대상 |
|----------------|------------|-------------|
| **User** | - | - |
| **Host** | HostUser | User(user_id) |
| **Event** | EventDetailImage | Host(host_id) |
| **TicketItem** | ItemOptionGroup | Event(event_id) |
| **OptionGroup** | Option | Event(event_id) |
| **Cart** | CartLineItem, CartOptionAnswer | User(user_id), TicketItem(item_id) |
| **Order** | OrderLine, OrderOptionAnswer | User(user_id), Event(event_id) |
| **IssuedTicket** | IssuedTicketOptionAnswer | User(user_id), Event(event_id), TicketItem(ticket_item_id), OrderLine(order_line_id) |
| **CouponCampaign** | IssuedCoupon | User(user_id) |
| **EventSettlement** | - | Event(event_id) |
| **TransactionSettlement** | FeeDetail | Event(event_id) |
| **Comment** | - | User(user_id), Event(event_id) |

## DDL 요약 (핵심 테이블)

### tbl_user
```sql
CREATE TABLE tbl_user (
  user_id BIGINT AUTO_INCREMENT PRIMARY KEY,
  created_at DATETIME, updated_at DATETIME,
  account_role VARCHAR(255),      -- SUPER_ADMIN/ADMIN/MANAGER/USER
  account_state VARCHAR(255),     -- NORMAL/DELETED/FORBIDDEN/SUSPENDED
  last_login_at DATETIME,
  marketing_agree BIT, receive_mail BIT,
  oid VARCHAR(255),               -- 카카오 고유 ID
  provider VARCHAR(255),          -- KAKAO
  email VARCHAR(255), name VARCHAR(255),
  phone_number VARCHAR(255), image_key VARCHAR(255),
  UNIQUE KEY (oid, provider)
);
```

### tbl_event
```sql
CREATE TABLE tbl_event (
  event_id BIGINT AUTO_INCREMENT PRIMARY KEY,
  created_at DATETIME, updated_at DATETIME,
  name VARCHAR(30),
  start_at DATETIME, run_time BIGINT,
  content LONGTEXT, image_key VARCHAR(255),
  place_name VARCHAR(255), place_address VARCHAR(255),
  latitude DOUBLE, longitude DOUBLE,
  host_id BIGINT,                 -- ID 참조 (FK 없음)
  status VARCHAR(255)             -- PREPARING/OPEN/CALCULATING/CLOSED/DELETED
);
```

### tbl_ticket_item
```sql
CREATE TABLE tbl_ticket_item (
  ticket_item_id BIGINT AUTO_INCREMENT PRIMARY KEY,
  created_at DATETIME, updated_at DATETIME,
  event_id BIGINT,                -- FK → tbl_event
  name VARCHAR(255), description VARCHAR(255),
  pay_type VARCHAR(255),          -- FREE_TICKET/PRICE_TICKET/DUDOONG_TICKET
  type VARCHAR(255),              -- FIRST_COME_FIRST_SERVED/APPROVAL
  amount VARCHAR(255) NOT NULL,   -- 가격 (Money VO)
  quantity BIGINT,                -- 총 수량
  supply_count BIGINT,            -- 남은 수량
  purchase_limit BIGINT,          -- 1인당 구매 제한
  sale_start_at DATETIME, sale_end_at DATETIME,
  is_sellable BIT, is_quantity_public BIT,
  ticket_item_status VARCHAR(255) DEFAULT 'VALID',
  account_holder VARCHAR(255), account_number VARCHAR(255), bank_name VARCHAR(255),
  FOREIGN KEY (event_id) REFERENCES tbl_event(event_id)
);
```

### tbl_order
```sql
CREATE TABLE tbl_order (
  order_id BIGINT AUTO_INCREMENT PRIMARY KEY,
  created_at DATETIME, updated_at DATETIME,
  uuid VARCHAR(255) NOT NULL,     -- 외부 식별자
  order_no VARCHAR(255),          -- "R" + (1000000 + order_id)
  order_name VARCHAR(255),
  order_status VARCHAR(255) NOT NULL,  -- 9개 상태
  order_method VARCHAR(255) NOT NULL,  -- APPROVAL/PAYMENT
  user_id BIGINT NOT NULL,
  event_id BIGINT NOT NULL,
  coupon_id BIGINT,
  amount VARCHAR(255) NOT NULL,        -- 총 금액
  discount_amount VARCHAR(255),
  supply_amount VARCHAR(255),
  vat_amount VARCHAR(255),
  payment_amount VARCHAR(255),         -- 실 결제 금액
  payment_key VARCHAR(255),            -- PG 결제 키
  payment_method VARCHAR(255),
  payment_provider VARCHAR(255),
  receipt_url VARCHAR(255),
  coupon_name VARCHAR(255),
  approved_at DATETIME,
  with_draw_at DATETIME
);
```

### tbl_issued_ticket
```sql
CREATE TABLE tbl_issued_ticket (
  issued_ticket_id BIGINT AUTO_INCREMENT PRIMARY KEY,
  created_at DATETIME, updated_at DATETIME,
  uuid VARCHAR(255) NOT NULL,          -- QR 코드 식별자
  issued_ticket_no VARCHAR(255),
  issued_ticket_status VARCHAR(255),   -- ENTRANCE_INCOMPLETE/ENTRANCE_COMPLETED/CANCELED
  event_id BIGINT, ticket_item_id BIGINT,
  order_line_id BIGINT, order_uuid VARCHAR(255),
  user_id BIGINT, user_name VARCHAR(255),
  email VARCHAR(255), phone_number VARCHAR(255),
  ticket_name VARCHAR(255), ticket_type VARCHAR(255),
  pay_type VARCHAR(255),
  amount VARCHAR(255) NOT NULL,
  entered_at DATETIME
);
```

### tbl_host / tbl_host_user
```sql
CREATE TABLE tbl_host (
  host_id BIGINT AUTO_INCREMENT PRIMARY KEY,
  created_at DATETIME, updated_at DATETIME,
  master_user_id BIGINT,          -- 방장 유저
  partner BIT,                    -- 제휴 호스트 여부
  name VARCHAR(15), introduce VARCHAR(255),
  contact_email VARCHAR(255), contact_number VARCHAR(15),
  image_key VARCHAR(255),
  slack_url VARCHAR(255)          -- Slack 웹훅 URL
);

CREATE TABLE tbl_host_user (
  host_user_id BIGINT AUTO_INCREMENT PRIMARY KEY,
  created_at DATETIME, updated_at DATETIME,
  host_id BIGINT,                 -- FK → tbl_host
  user_id BIGINT,
  role VARCHAR(255),              -- MASTER/MANAGER/GUEST
  active BIT,                     -- 초대 승인 여부
  UNIQUE KEY (host_id, user_id)
);
```

### tbl_event_settlement / tbl_transaction_settlement
```sql
CREATE TABLE tbl_event_settlement (
  event_settlement_id BIGINT AUTO_INCREMENT PRIMARY KEY,
  event_id BIGINT,
  event_settlement_status VARCHAR(255),  -- READY/CALCULATED/ADMIN_CONFIRM/CLIENT_CONFIRM
  total_sales_amount VARCHAR(255),
  total_amount VARCHAR(255),
  payment_amount VARCHAR(255),
  pg_fee VARCHAR(255), pg_fee_vat VARCHAR(255),
  dudoong_amount VARCHAR(255), dudoong_fee VARCHAR(255),
  coupon_amount VARCHAR(255),
  event_order_excel_key VARCHAR(255)     -- S3 엑셀 키
);

CREATE TABLE tbl_transaction_settlement (
  transaction_settlement_id BIGINT AUTO_INCREMENT PRIMARY KEY,
  event_id BIGINT,
  order_uuid VARCHAR(255),
  payment_key VARCHAR(255), payment_method VARCHAR(255),
  payment_amount VARCHAR(255),
  settlement_amount VARCHAR(255),
  fee_supply_amount VARCHAR(255), fee_vat VARCHAR(255),
  interest_fee VARCHAR(255),
  approved_at DATETIME, sold_date DATE, paid_out_date DATE,
  transaction_key VARCHAR(255)
);
```
